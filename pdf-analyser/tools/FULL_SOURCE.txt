name: GPU integration smoke test

on:
  workflow_dispatch:

permissions:
  contents: read

# This workflow requires a self-hosted runner with labels:
# self-hosted, linux, x64, gpu
jobs:
  smoke:
    runs-on: [self-hosted, linux, x64, gpu]
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v4

      - name: Verify host GPU
        run: |
          nvidia-smi
          docker run --rm --gpus all nvidia/cuda:12.6.3-base-ubuntu24.04 nvidia-smi

      - name: Start GPU stack
        run: |
          cp .env.example .env
          docker compose -f docker-compose.yml -f docker-compose.gpu.yml up -d --build

      - name: Verify CUDA in extractor
        run: |
          docker compose -f docker-compose.yml -f docker-compose.gpu.yml exec -T extractor-api \
            python -c 'import torch; assert torch.cuda.is_available(); print({"torch": torch.__version__, "gpu": torch.cuda.get_device_name(0), "capability": torch.cuda.get_device_capability(0), "arch": torch.cuda.get_arch_list()})'

      - name: Run extraction smoke test
        run: ./scripts/smoke_test.sh tests/fixtures/scientific_sample.pdf

      - name: Collect logs on failure
        if: failure()
        run: docker compose -f docker-compose.yml -f docker-compose.gpu.yml logs --no-color --tail=500

      - name: Stop stack
        if: always()
        run: docker compose -f docker-compose.yml -f docker-compose.gpu.yml down --remove-orphans

name: Publish images to GHCR

on:
  push:
    tags:
      - "v*"

permissions:
  contents: read
  packages: write

jobs:
  publish:
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        include:
          - variant: cpu
            torch_index: https://download.pytorch.org/whl/cpu
            suffix: cpu
          - variant: gpu-cu126
            torch_index: https://download.pytorch.org/whl/cu126
            suffix: gpu-cu126

    steps:
      - uses: actions/checkout@v4

      - uses: docker/setup-buildx-action@v3

      - uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Normalize repository owner
        id: repo
        shell: bash
        run: echo "owner=${GITHUB_REPOSITORY_OWNER,,}" >> "$GITHUB_OUTPUT"

      - uses: docker/metadata-action@v5
        id: meta
        with:
          images: ghcr.io/${{ steps.repo.outputs.owner }}/pdf-extractor-api
          tags: |
            type=ref,event=tag,suffix=-${{ matrix.suffix }}
            type=raw,value=${{ matrix.suffix }}

      - uses: docker/build-push-action@v6
        with:
          context: ./api
          file: ./api/Dockerfile
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}
          build-args: |
            TORCH_INDEX_URL=${{ matrix.torch_index }}
            TORCH_VERSION=2.11.0
            TORCHVISION_VERSION=0.26.0
          cache-from: type=gha,scope=publish-${{ matrix.variant }}
          cache-to: type=gha,mode=max,scope=publish-${{ matrix.variant }}

# Contributing

## Development checks

Create a Python 3.12 virtual environment and install lightweight development dependencies:

```bash
python3.12 -m venv .venv
source .venv/bin/activate
python -m pip install --upgrade pip
python -m pip install -r api/requirements-dev.txt
```

Before opening a pull request:

```bash
make format
make lint
make test

docker compose -f docker-compose.yml config --quiet
docker compose -f docker-compose.yml -f docker-compose.gpu.yml config --quiet
```

For extraction changes, run the smoke test against both CPU and GPU when hardware is available. Record any changes in table structure, formulas, image crops, performance, and memory use against the project benchmark PDFs.

Do not commit:

- `.env`
- user PDFs
- extracted assets
- API results containing article content
- model caches
- credentials or organization-specific endpoints

SHELL := /bin/bash
CPU_COMPOSE := docker compose -f docker-compose.yml
GPU_COMPOSE := docker compose -f docker-compose.yml -f docker-compose.gpu.yml

.PHONY: init build-cpu up-cpu build-gpu up-gpu down down-gpu logs ps ready runtime test lint format smoke clean

init:
	cp -n .env.example .env || true
	mkdir -p data/assets output

build-cpu:
	$(CPU_COMPOSE) build extractor-api

up-cpu: init
	$(CPU_COMPOSE) up -d --build

build-gpu:
	$(GPU_COMPOSE) build extractor-api

up-gpu: init
	$(GPU_COMPOSE) up -d --build

# Use down-gpu after a GPU launch so Compose sees the same merged project definition.
down:
	$(CPU_COMPOSE) down --remove-orphans

down-gpu:
	$(GPU_COMPOSE) down --remove-orphans

logs:
	$(CPU_COMPOSE) logs -f --tail=200

ps:
	$(CPU_COMPOSE) ps

ready:
	curl -fsS http://localhost:$${API_PORT:-8090}/health/ready | python -m json.tool

runtime:
	curl -fsS http://localhost:$${API_PORT:-8090}/v1/runtime | python -m json.tool

test:
	cd api && python -m pytest

lint:
	cd api && ruff check . && ruff format --check .

format:
	cd api && ruff check --fix . && ruff format .

smoke:
	./scripts/smoke_test.sh tests/fixtures/scientific_sample.pdf

clean:
	$(GPU_COMPOSE) down -v --remove-orphans
	rm -rf output/* data/assets/*
	touch data/assets/.gitkeep

# Scientific PDF Extraction Stack

Production-oriented first-stage extraction for scientific articles using **Docling** and **GROBID**.

One HTTP request processes a PDF with both extractors in parallel and returns:

- Docling document JSON
- Docling Markdown
- Docling formulas with provenance
- Docling structured-table summaries and explicit failure flags
- Persisted figure crops as PNG
- Persisted table crops as PNG
- GROBID TEI XML
- Parsed GROBID metadata, sections, references, figures/tables, and formulas
- A conservative merged preview that does not hide parser disagreements

The repository supports two runtime modes from the same codebase:

- **CPU:** portable default; no GPU is required
- **NVIDIA GPU:** CUDA-accelerated Docling with a separate Compose override

> This service is the extraction layer. It intentionally does not perform the final canonical merge, quality scoring, agentic repair, chunking, embedding, or OpenSearch indexing. Those should consume this service's output as later pipeline stages.

---

## 1. Architecture

```text
                              PDF upload
                                   |
                                   v
                    FastAPI /v1/extract endpoint
                                   |
                  +----------------+----------------+
                  |                                 |
                  v                                 v
      Docling in extractor-api               GROBID REST service
      - layout and reading order              - scholarly metadata
      - OCR                                    - article sections
      - accurate table structure              - citations/references
      - formula enrichment                    - figure/table/formula TEI
      - figure/table image crops              - PDF coordinates
                  |                                 |
                  +----------------+----------------+
                                   |
                                   v
                         Combined JSON response
                                   |
                                   +--> persisted assets
                                        data/assets/<sha256>/
                                        ├── figures/*.png
                                        └── tables/*.png
```

Docling and GROBID outputs remain separately available. The `merged` field is deliberately conservative so that later quality checks can compare the parsers rather than losing information through an early merge.

## 3. What to install before using the project

### 3.1 Required on every machine

Recommended host:

- Linux x86_64
- 16 GB RAM or more
- 25–40 GB free disk space for Docker images and model caches
- Several CPU cores; accurate OCR/table/formula processing is compute-intensive

Install:

1. **Git**
2. **Docker Engine**
3. **Docker Compose v2 plugin**
4. `curl`, `jq`, and `make` for the documented commands

On Ubuntu/Debian, the small command-line tools can be installed with:

```bash
sudo apt-get update
sudo apt-get install -y git curl jq make
```

Install Docker Engine and the Compose plugin using Docker's official installation guide:

- https://docs.docker.com/engine/install/
- https://docs.docker.com/compose/install/linux/

Verify:

```bash
docker --version
docker compose version
```

You do **not** need to install Python, Docling, GROBID, Java, PyTorch, or the CUDA toolkit directly on the host. They are provided by the containers.

### 3.2 Additional requirements for NVIDIA GPU mode

GPU mode additionally requires:

1. A supported NVIDIA GPU
2. A working NVIDIA host driver
3. NVIDIA Container Toolkit configured for Docker

Verify the host driver:

```bash
nvidia-smi
```

Install NVIDIA Container Toolkit using NVIDIA's current official guide:

- https://docs.nvidia.com/datacenter/cloud-native/container-toolkit/latest/install-guide.html

After installation, configure and restart Docker:

```bash
sudo nvidia-ctk runtime configure --runtime=docker
sudo systemctl restart docker
```

Verify container access:

```bash
docker run --rm --gpus all \
  nvidia/cuda:12.6.3-base-ubuntu24.04 \
  nvidia-smi
```

The repository uses the PyTorch **CUDA 12.6** wheel in GPU mode. This is intentional because it supports Pascal GPUs such as the GTX 1080 (`compute capability 6.1`) while also supporting newer architectures in the tested wheel. Do not switch this project to the CUDA 12.8 wheel on a GTX 1080; current CUDA 12.8 PyTorch builds may omit the required Pascal kernels.

## 4. First-time setup

Clone the repository:

```bash
git clone https://github.com/YOUR_ORG/YOUR_REPOSITORY.git
cd YOUR_REPOSITORY
```

Create the local environment file and runtime directories:

```bash
make init
```

Equivalent manual commands:

```bash
cp .env.example .env
mkdir -p data/assets output
```

### Important: update the GROBID contact email

Before public or production use, edit:

```text
grobid.yaml
```

Replace:

```yaml
mailto: "replace-with-your-email@example.com"
```

with a monitored contact email. This is used only if Crossref consolidation is enabled. The supplied configuration uses Glutton and the extraction endpoint defaults consolidation to `0`, so ordinary extraction does not require Glutton or Crossref.

## 5. Start in CPU mode

CPU mode is the portable default and works on a machine without an NVIDIA GPU.

Build and start:

```bash
make up-cpu
```

Equivalent Compose command:

```bash
docker compose \
  -f docker-compose.yml \
  up -d --build
```

Follow logs:

```bash
docker compose logs -f --tail=200
```

The first startup can take several minutes because:

- the full GROBID image is large
- GROBID preloads models according to `grobid.yaml`
- Docling/RapidOCR/Hugging Face model files are downloaded into the persistent model cache

Check readiness:

```bash
curl -fsS http://localhost:8090/health/ready | jq
```

Expected:

```json
{
  "status": "ready",
  "grobid": true,
  "docling": true
}
```

Inspect runtime details:

```bash
curl -fsS http://localhost:8090/v1/runtime | jq
```

In CPU mode you should see approximately:

```json
{
  "docling": {
    "configured_device": "cpu",
    "ocr_backend": "onnxruntime",
    "cuda_available": false
  }
}
```

## 6. Start in NVIDIA GPU mode

First verify GPU access:

```bash
./scripts/check_gpu.sh
```

Build and start with both Compose files:

```bash
make up-gpu
```

Equivalent command:

```bash
docker compose \
  -f docker-compose.yml \
  -f docker-compose.gpu.yml \
  up -d --build
```

The GPU override changes only the Docling service:

```text
PyTorch wheel:        CUDA 12.6
Docling device:       cuda
RapidOCR backend:     torch
GROBID:               CPU
Accuracy settings:    unchanged
```

Verify CUDA inside the running extractor:

```bash
docker compose \
  -f docker-compose.yml \
  -f docker-compose.gpu.yml \
  exec -T extractor-api \
  python -c '
import torch
print({
    "torch": torch.__version__,
    "cuda_available": torch.cuda.is_available(),
    "cuda_version": torch.version.cuda,
    "gpu": torch.cuda.get_device_name(0) if torch.cuda.is_available() else None,
    "capability": torch.cuda.get_device_capability(0) if torch.cuda.is_available() else None,
    "compiled_architectures": torch.cuda.get_arch_list() if torch.cuda.is_available() else [],
})
'
```

For a GTX 1080, a healthy result looks similar to:

```json
{
  "torch": "2.11.0+cu126",
  "cuda_available": true,
  "cuda_version": "12.6",
  "gpu": "NVIDIA GeForce GTX 1080",
  "capability": [6, 1],
  "compiled_architectures": [
    "sm_50",
    "sm_60",
    "sm_70",
    "sm_75",
    "sm_80",
    "sm_86",
    "sm_90"
  ]
}
```

Monitor GPU use during extraction:

```bash
watch -n 1 nvidia-smi
```

### Switching between GPU and CPU

Stop with the same Compose definition used to start the stack:

```bash
make down-gpu
```

Then start CPU mode:

```bash
make up-cpu
```

The CPU and GPU extractor images have different local names, so both may remain cached.

## 7. Service URLs

Default local endpoints:

| Service | URL |
|---|---|
| Extraction API | `http://localhost:8090` |
| Swagger UI | `http://localhost:8090/docs` |
| OpenAPI JSON | `http://localhost:8090/openapi.json` |
| Liveness | `http://localhost:8090/health/live` |
| Readiness | `http://localhost:8090/health/ready` |
| Runtime/GPU diagnostics | `http://localhost:8090/v1/runtime` |
| GROBID API | `http://127.0.0.1:8086` |
| GROBID admin | `http://127.0.0.1:8085` |
| Extracted assets | `http://localhost:8090/assets/...` |

GROBID is bound to localhost by default. The extraction API is bound to all host interfaces by default; change `API_BIND_HOST` in `.env` if needed.

## 8. Process a PDF

Run:

```bash
curl --fail --show-error \
  -X POST "http://localhost:8090/v1/extract" \
  -H "accept: application/json" \
  -F "file=@./article.pdf;type=application/pdf" \
  -F "include_docling_markdown=true" \
  -F "include_docling_json=true" \
  -F "include_grobid_tei=true" \
  -F "export_picture_images=true" \
  -F "export_table_images=true" \
  -F "grobid_consolidate_header=0" \
  -F "grobid_consolidate_citations=0" \
  -o result.json
```

Pretty-print the result:

```bash
jq . result.json > result.pretty.json
```

or:

```bash
python -m json.tool result.json > result.pretty.json
```

For large JSON files, inspect only the useful sections:

```bash
jq '.docling.stats' result.json
jq '.docling.parsed.formulas' result.json
jq '.docling.parsed.tables' result.json
jq '.docling.parsed.quality_flags' result.json
jq '.grobid.stats' result.json
jq '.grobid.parsed.metadata' result.json
jq '.grobid.parsed.formulas' result.json
```

Run the included end-to-end smoke script:

```bash
make smoke
```

It uses `tests/fixtures/scientific_sample.pdf` and writes results to `output/`.

```json
    "raw": {
      "tei_xml": "..."
    },
    "parsed": {
      "metadata": {},
      "sections": [],
      "references": [],
      "figures_and_tables": [],
      "formulas": []
    }
  },
  "merged": {
    "merge_status": "preview_only"
  },
  "warnings": []
}
```

### Partial success

If one parser fails but the other succeeds, the endpoint returns HTTP 200 with:

```json
{
  "status": "partial",
  "docling": {
    "status": "completed"
  },
  "grobid": {
    "status": "failed",
    "error": "..."
  }
}
```

If both fail, the endpoint returns HTTP 502.

## 10. Figure and table image persistence

Docling image generation and image saving are separate operations. This repository performs both.

The relevant defaults are:

```env
DOCLING_GENERATE_PAGE_IMAGES=true
DOCLING_GENERATE_PICTURE_IMAGES=true
DOCLING_IMAGES_SCALE=2.0
DOCLING_EXPORT_PICTURES=true
DOCLING_EXPORT_TABLES=true
```

Images are stored on the host:

```text
data/assets/<pdf-sha256>/
├── figures/
│   ├── figure_001.png
│   ├── figure_002.png
│   └── ...
└── tables/
    ├── table_001.png
    ├── table_002.png
    └── ...
```

The JSON contains relative HTTP URLs such as:

```json
{
  "relative_path": "<sha256>/figures/figure_001.png",
  "url": "/assets/<sha256>/figures/figure_001.png",
  "width": 1600,
  "height": 900,
  "size_bytes": 248123
}
```

Open locally:

```text
http://localhost:8090/assets/<sha256>/figures/figure_001.png
```

### Image quality

`DOCLING_IMAGES_SCALE=2.0` is the recommended starting point. Approximate behavior:

- `1.0`: smaller/faster; roughly standard PDF rendering scale
- `2.0`: good balance for charts, diagrams, and labels
- `3.0`: larger and slower; useful for very small labels
- `4.0`: high memory/storage cost; use only after measurement

Increasing scale enlarges vector/rendered content. It cannot recover detail missing from a low-resolution raster image embedded in the original PDF.

## 11. Formula output

Docling formulas are available under:

```text
docling.parsed.formulas
```

Each item contains:

- Docling ID
- enriched formula text
- original recognized text
- page/bounding-box provenance

Example:

```json
{
  "id": "#/texts/47",
  "text": "\\mathcal{L}=-\\sum_i y_i\\log(p_i)",
  "original": "L = - Σ yᵢ log(pᵢ)",
  "provenance": [
    {
      "page_no": 5,
      "bbox": {}
    }
  ]
}
```

GROBID formulas are available under:

```text
grobid.parsed.formulas
```

GROBID is useful for equation labels/numbers, TEI structure, and coordinates. Docling is generally more useful for enriched mathematical text. Final formula alignment belongs in the later canonical merge/quality stage.

## 12. Table output and early quality flags

Docling tables are available in full under:

```text
docling.raw.document_json.tables
```

A compact summary is available under:

```text
docling.parsed.tables
```

Example successful table:

```json
{
  "rows": 5,
  "columns": 4,
  "cell_count": 20,
  "structure_status": "structured"
}
```

A layout region can be detected as a table while structure reconstruction fails. This looks like:

```json
{
  "rows": 0,
  "columns": 0,
  "cell_count": 0,
  "structure_status": "empty_structure"
}
```

The service adds a quality flag:

```json
{
  "type": "table_empty_structure",
  "table_id": "#/tables/0",
  "index": 1,
  "provenance": []
}
```

This is not the final quality gate, but it is a useful deterministic signal for the next stage. Real PDFs vary greatly; some tables will reconstruct correctly and others will require a fallback or later agentic repair.

To experiment with cell matching:

```env
DOCLING_TABLE_DO_CELL_MATCHING=true
```

For a problematic PDF, try:

```env
DOCLING_TABLE_DO_CELL_MATCHING=false
```

Recreate the extractor and compare results. Do not assume one setting is universally best; evaluate against your PDF benchmark set.

## 17. Production deployment guidance

The stack is production-oriented, but a real high-volume deployment should add infrastructure around it.

### Recommended before internet exposure

- Put the API behind a reverse proxy or API gateway.
- Add authentication and authorization.
- Add TLS.
- Set request-size and request-timeout limits at the proxy.
- Restrict or disable public asset serving if documents are sensitive.
- Do not expose GROBID publicly; keep it on the private Docker network.
- Add malware scanning and PDF security controls for untrusted uploads.

### Use asynchronous jobs for large workloads

The current endpoint is synchronous: the HTTP request remains open until both parsers finish. This is convenient for development and low throughput.

For production ingestion, prefer:

```text
POST /jobs
    -> store PDF
    -> enqueue extraction
    -> return job ID

worker
    -> Docling + GROBID
    -> persist JSON/assets
    -> update status

GET /jobs/<id>
    -> status/result
```

Use a queue such as Redis/RQ, Celery, RabbitMQ, Kafka, or your existing job system.

### Model memory and scaling

The API intentionally starts one Uvicorn worker:

```text
--workers 1
```

Every worker would load another copy of Docling/PyTorch models. To scale throughput:

- keep one worker per container
- run multiple extractor containers
- assign one GPU or controlled GPU share per container
- keep `DOCLING_MAX_CONCURRENCY=1` until memory and throughput are measured

### Storage

The local bind mount is suitable for development and a single node. For production:

- store original PDFs in object storage
- store figure/table images in object storage
- store extraction JSON in object storage or a document database
- store only URLs, hashes, metadata, chunks, and vectors in OpenSearch
- implement retention and deletion policies

### Observability

Add:

- structured JSON logging
- request/job correlation IDs
- Prometheus metrics
- extraction durations by parser
- failure counts by stage
- GPU memory/utilization monitoring
- model version/configuration metadata in every result

### Deduplication

The API calculates SHA-256 while streaming the upload. The same hash is used as the asset namespace. Use that hash as a stable document identity and deduplication key in the ingestion layer.

## 18. GROBID consolidation and Glutton

The supplied `grobid.yaml` points Glutton to:

```text
http://host.docker.internal:8080
```

Normal extraction sends:

```text
consolidateHeader=0
consolidateCitations=0
```

Therefore Glutton is not required for the normal path.

Only enable consolidation when Glutton is running and reachable:

```bash
-F "grobid_consolidate_header=1" \
-F "grobid_consolidate_citations=1"
```

External consolidation can increase latency and introduces an external dependency. Keep it disabled in deterministic extraction benchmarks unless the benchmark explicitly includes consolidation.

## 19. Troubleshooting

### `could not select device driver "nvidia"`

Cause: Docker cannot access the NVIDIA runtime.

Check:

```bash
nvidia-smi
docker run --rm --gpus all nvidia/cuda:12.6.3-base-ubuntu24.04 nvidia-smi
```

Install/configure NVIDIA Container Toolkit, then restart Docker.

### `no kernel image is available for execution on the device`

Cause: the PyTorch wheel does not include kernels compatible with the GPU architecture.

For a GTX 1080 (`CC 6.1`), use this repository's CUDA 12.6 build. Verify:

```bash
docker compose \
  -f docker-compose.yml \
  -f docker-compose.gpu.yml \
  exec -T extractor-api \
  python -c 'import torch; print(torch.__version__, torch.version.cuda, torch.cuda.get_arch_list())'
```

You should see `+cu126` and an architecture list that includes the Pascal major family (`sm_60`).

### RapidOCR permission denied in `site-packages/rapidocr/models`

The Dockerfile makes the RapidOCR model directory writable by the non-root application user. The entrypoint also fixes mounted asset/model directory ownership.

If migrating an old root-owned model volume, remove it and allow the repository to create a fresh one:

```bash
docker compose \
  -f docker-compose.yml \
  -f docker-compose.gpu.yml \
  down -v
```

Then rebuild/start.

### Pydantic `grobid_tei_coordinates` JSON decoding error

This repository uses `NoDecode` and a comma-splitting validator. Keep the environment value as:

```env
GROBID_TEI_COORDINATES=persName,figure,ref,biblStruct,formula,s,p,title,affiliation,note
```

### Images are detected but no PNG files appear

Check:

```bash
curl -fsS http://localhost:8090/v1/runtime | jq
find data/assets -type f -name '*.png'
```

Confirm:

```env
DOCLING_GENERATE_PAGE_IMAGES=true
DOCLING_GENERATE_PICTURE_IMAGES=true
DOCLING_EXPORT_PICTURES=true
DOCLING_EXPORT_TABLES=true
```

The code must call `PictureItem.get_image()` / `TableItem.get_image()` and save the returned images. This repository does that in `api/app/services/docling_service.py`.

### Exported images are too small

Increase:

```env
DOCLING_IMAGES_SCALE=3.0
```

Then recreate the extractor. Higher values increase runtime, memory use, and storage.

### Table object has zero rows, columns, and cells

This means the layout stage detected a table region but the structure stage did not reconstruct a usable grid.

Inspect:

```bash
jq '.docling.parsed.tables' result.json
jq '.docling.parsed.quality_flags' result.json
```

Try changing:

```env
DOCLING_TABLE_DO_CELL_MATCHING=false
```

and compare results. Keep this as a quality-gate failure for later specialized fallback/agentic repair.

### Extraction takes a long time

Accurate table mode, OCR, formula enrichment, and image rendering are all enabled. First-run model downloads are also slow.

Inspect:

```bash
docker stats
curl -fsS http://localhost:8090/v1/runtime | jq
```

GPU mode should show a Python process and utilization in:

```bash
watch -n 1 nvidia-smi
```

Do not increase Uvicorn workers to improve speed; that duplicates model memory. Scale containers after benchmarking.

### Old container-name conflicts

This repository does not set `container_name`, so independent Compose projects do not compete for global fixed names. If old manually named containers remain from a previous prototype:

```bash
docker ps -a
docker rm -f grobid pdf-extractor-api
```

### Compose warns that `version` is obsolete

This repository does not include the legacy top-level `version:` key.

## 20. Upgrade procedure

Do not update Docling, PyTorch, CUDA wheels, FastAPI, or GROBID blindly in production.

Recommended process:

1. Create a branch.
2. Update one dependency family at a time.
3. Build CPU and GPU images.
4. Run unit tests.
5. Run the GPU smoke workflow.
6. Process your manually reviewed benchmark PDF set.
7. Compare:
   - page/text counts
   - table rows/columns/cells
   - formula outputs
   - figure/table crops
   - extraction time
   - GPU memory
8. Record the new parser/model versions.
9. Deploy to staging before production.

Your benchmark should include simple born-digital articles, two-column layouts, scanned pages, complex/multi-page tables, equations, charts, diagrams, and low-quality PDFs.

from functools import lru_cache
from pathlib import Path
from typing import Annotated, Literal

from pydantic import Field, field_validator
from pydantic_settings import BaseSettings, NoDecode, SettingsConfigDict


class Settings(BaseSettings):
    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        case_sensitive=False,
        extra="ignore",
    )

    app_name: str = "Scientific PDF Extraction API"
    app_version: str = "0.2.0"
    log_level: str = "INFO"

    max_upload_size_mb: int = Field(default=100, gt=0)
    serve_assets: bool = True
    assets_dir: Path = Path("/data/assets")
    assets_url_prefix: str = "/assets"

    grobid_base_url: str = "http://grobid:8070"
    grobid_timeout_seconds: float = Field(default=300.0, gt=0)
    grobid_max_retries: int = Field(default=2, ge=1, le=10)
    grobid_tei_coordinates: Annotated[list[str], NoDecode] = Field(
        default_factory=lambda: [
            "persName",
            "figure",
            "ref",
            "biblStruct",
            "formula",
            "s",
            "p",
            "title",
            "affiliation",
            "note",
        ]
    )

    docling_preload_on_startup: bool = True
    docling_max_num_pages: int = Field(default=300, gt=0)
    docling_max_concurrency: int = Field(default=1, gt=0, le=8)
    docling_document_timeout_seconds: float = Field(default=1800.0, gt=0)
    docling_accelerator_device: str = "cpu"
    docling_accelerator_num_threads: int = Field(default=4, gt=0, le=64)

    docling_ocr_backend: Literal["onnxruntime", "torch"] = "onnxruntime"
    docling_do_ocr: bool = True
    docling_ocr_force_full_page: bool = False
    docling_ocr_bitmap_area_threshold: float = Field(default=0.05, ge=0.0, le=1.0)

    docling_do_table_structure: bool = True
    docling_table_mode: Literal["fast", "accurate"] = "accurate"
    docling_table_do_cell_matching: bool = True
    docling_do_formula_enrichment: bool = True

    docling_generate_page_images: bool = True
    docling_generate_picture_images: bool = True
    docling_images_scale: float = Field(default=2.0, ge=0.5, le=4.0)
    docling_export_pictures: bool = True
    docling_export_tables: bool = True

    @field_validator("grobid_base_url")
    @classmethod
    def strip_trailing_slash(cls, value: str) -> str:
        return value.rstrip("/")

    @field_validator("assets_url_prefix")
    @classmethod
    def normalize_assets_prefix(cls, value: str) -> str:
        cleaned = "/" + value.strip("/")
        return cleaned if cleaned != "/" else "/assets"

    @field_validator("grobid_tei_coordinates", mode="before")
    @classmethod
    def parse_coordinate_list(cls, value: object) -> object:
        if isinstance(value, str):
            return [item.strip() for item in value.split(",") if item.strip()]
        return value

    @field_validator("docling_accelerator_device")
    @classmethod
    def validate_accelerator_device(cls, value: str) -> str:
        normalized = value.strip().lower()
        if normalized in {"auto", "cpu", "cuda"}:
            return normalized
        if normalized.startswith("cuda:") and normalized[5:].isdigit():
            return normalized
        raise ValueError("Expected auto, cpu, cuda, or cuda:<index>.")

    @property
    def max_upload_size_bytes(self) -> int:
        return self.max_upload_size_mb * 1024 * 1024


@lru_cache
def get_settings() -> Settings:
    return Settings()

from app.config import get_settings
from app.models import ExtractionResponse, ParserResult
from app.services.docling_service import DoclingService
from app.services.grobid_service import GrobidService
from app.services.merge_service import build_merged_preview

settings = get_settings()
logging.basicConfig(
    level=getattr(logging, settings.log_level.upper(), logging.INFO),
    format="%(asctime)s %(levelname)s %(name)s %(message)s",
)
logger = logging.getLogger(__name__)


@asynccontextmanager
async def lifespan(app: FastAPI):
    app.state.docling = DoclingService(settings)
    app.state.grobid = GrobidService(settings)

    if settings.docling_preload_on_startup:
        try:
            await app.state.docling.initialize()
        except Exception:
            logger.exception("Docling initialization failed during startup.")

    yield
    await app.state.grobid.close()


app = FastAPI(
    title=settings.app_name,
    version=settings.app_version,
    description=(
        "Upload a scientific PDF and receive Docling JSON/Markdown, persisted figure/table "
        "assets, GROBID TEI, parsed views, and a conservative merged preview."
    ),
    lifespan=lifespan,
)

if settings.serve_assets:
    settings.assets_dir.mkdir(parents=True, exist_ok=True)
    app.mount(
        settings.assets_url_prefix,
        StaticFiles(directory=str(settings.assets_dir)),
        name="assets",
    )


@app.get("/health/live")
async def live() -> dict[str, str]:
    return {"status": "ok"}


@app.get("/health/ready")
async def ready(request: Request) -> dict[str, Any]:
    grobid_alive = await request.app.state.grobid.is_alive()
    docling_ready = request.app.state.docling.is_initialized

    if not grobid_alive or not docling_ready:
        raise HTTPException(
            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
            detail={
                "status": "not_ready",
                "grobid": grobid_alive,
                "docling": docling_ready,
                "docling_error": request.app.state.docling.initialization_error,
            },
        )

    return {"status": "ready", "grobid": True, "docling": True}


@app.get("/v1/runtime")
async def runtime(request: Request) -> dict[str, Any]:
    return {
        "app": {"name": settings.app_name, "version": settings.app_version},
        "docling": await request.app.state.docling.runtime_info(),
        "grobid": {
            "alive": await request.app.state.grobid.is_alive(),
            "version": await request.app.state.grobid.version(),
        },
    }

    include_docling_json: Annotated[bool, Form()] = True,
    include_grobid_tei: Annotated[bool, Form()] = True,
    export_picture_images: Annotated[bool, Form()] = settings.docling_export_pictures,
    export_table_images: Annotated[bool, Form()] = settings.docling_export_tables,
    grobid_consolidate_header: Annotated[str, Form(pattern="^[0-3]$")] = "0",
    grobid_consolidate_citations: Annotated[str, Form(pattern="^[0-2]$")] = "0",
    grobid_include_raw_citations: Annotated[bool, Form()] = True,
    grobid_include_raw_affiliations: Annotated[bool, Form()] = True,
    grobid_segment_sentences: Annotated[bool, Form()] = True,
) -> ExtractionResponse:
    request_id = str(uuid.uuid4())
    filename = safe_filename(file.filename)
    started = time.perf_counter()
    temp_path: Path | None = None

    try:
        temp_path, sha256, size_bytes = await save_upload_to_temp(
            file,
            max_bytes=settings.max_upload_size_bytes,
        )

        docling_started = time.perf_counter()
        grobid_started = time.perf_counter()
        docling_task = asyncio.create_task(
            request.app.state.docling.extract(
                temp_path,
                document_id=sha256,
                include_markdown=include_docling_markdown,
                export_pictures=export_picture_images,
                export_tables=export_table_images,
            )
        )
        grobid_task = asyncio.create_task(
            request.app.state.grobid.extract(
                temp_path,
                filename=filename,
                consolidate_header=grobid_consolidate_header,
                consolidate_citations=grobid_consolidate_citations,
                include_raw_citations=grobid_include_raw_citations,
                include_raw_affiliations=grobid_include_raw_affiliations,
                segment_sentences=grobid_segment_sentences,
                include_tei_xml=include_grobid_tei,
            )
        )

        docling_outcome, grobid_outcome = await asyncio.gather(
            docling_task,
            grobid_task,
            return_exceptions=True,
        )

        warnings: list[str] = []
        docling_data: dict[str, Any] | None = None
        grobid_data: dict[str, Any] | None = None

        if isinstance(docling_outcome, Exception):
            logger.error(
                "Docling failed for request %s: %s",
                request_id,
                docling_outcome,
                exc_info=(
                    type(docling_outcome),
                    docling_outcome,
                    docling_outcome.__traceback__,
                ),
            )
            docling = ParserResult(
                status="failed",
                duration_seconds=round(time.perf_counter() - docling_started, 3),
                error=str(docling_outcome),
            )
            warnings.append("Docling extraction failed; GROBID output may still be available.")
        else:
            docling_data = docling_outcome
            raw = docling_data.get("raw")
            if raw is not None and not include_docling_json:
                raw.pop("document_json", None)
            docling = ParserResult(
                status="completed",
                version=docling_data.get("version"),
                duration_seconds=docling_data["duration_seconds"],
                stats=docling_data.get("stats", {}),
                raw=raw,
                parsed=docling_data.get("parsed"),
            )
            if docling_data.get("parsed", {}).get("quality_flags"):
                warnings.append(
                    "Docling detected one or more tables with an empty row/column/cell structure."
                )

        if isinstance(grobid_outcome, Exception):
            logger.error(
                "GROBID failed for request %s: %s",
                request_id,
                grobid_outcome,
                exc_info=(
                    type(grobid_outcome),
                    grobid_outcome,
                    grobid_outcome.__traceback__,
                ),
            )
            grobid = ParserResult(
                status="failed",
                duration_seconds=round(time.perf_counter() - grobid_started, 3),
                error=str(grobid_outcome),
            )
            warnings.append("GROBID extraction failed; Docling output may still be available.")
        else:
            grobid_data = grobid_outcome
            grobid = ParserResult(
                status="completed",
                version=grobid_data.get("version"),
                duration_seconds=grobid_data["duration_seconds"],
                stats=grobid_data.get("stats", {}),
                raw=grobid_data.get("raw"),
                parsed=grobid_data.get("parsed"),
            )

        completed_count = int(docling.status == "completed") + int(grobid.status == "completed")
        overall_status = (
            "completed" if completed_count == 2 else "partial" if completed_count == 1 else "failed"
        )

        if overall_status == "failed":
            raise HTTPException(
                status_code=status.HTTP_502_BAD_GATEWAY,
                detail={
                    "request_id": request_id,
                    "message": "Both extractors failed.",
                    "docling_error": docling.error,
                    "grobid_error": grobid.error,
                },
            )

        return ExtractionResponse(
            request_id=request_id,
            status=overall_status,
            document={
                "filename": filename,
                "content_type": file.content_type or "application/pdf",
                "size_bytes": size_bytes,
                "sha256": sha256,
                "asset_namespace": sha256,
            },
            options={
                "include_docling_markdown": include_docling_markdown,
                "include_docling_json": include_docling_json,
                "include_grobid_tei": include_grobid_tei,
                "export_picture_images": export_picture_images,
                "export_table_images": export_table_images,
                "grobid_consolidate_header": grobid_consolidate_header,
                "grobid_consolidate_citations": grobid_consolidate_citations,
                "grobid_include_raw_citations": grobid_include_raw_citations,
                "grobid_include_raw_affiliations": grobid_include_raw_affiliations,
                "grobid_segment_sentences": grobid_segment_sentences,
            },
            processing={
                "total_duration_seconds": round(time.perf_counter() - started, 3),
                "parallel_extractors": True,
            },
            docling=docling,
            grobid=grobid,
            merged=build_merged_preview(
                docling_result=docling_data,
                grobid_result=grobid_data,
            ),
            warnings=warnings,
        )
    finally:
        if temp_path is not None:
            temp_path.unlink(missing_ok=True)

from typing import Any


def _ref_value(value: object) -> str | None:
    if isinstance(value, str):
        return value
    if isinstance(value, dict):
        for key in ("$ref", "cref", "ref"):
            candidate = value.get(key)
            if isinstance(candidate, str):
                return candidate
    return None


def _resolve_ref(document: dict[str, Any], reference: object) -> dict[str, Any] | None:
    value = _ref_value(reference)
    if not value or not value.startswith("#/"):
        return None

    node: object = document
    for part in value[2:].split("/"):
        if isinstance(node, dict):
            node = node.get(part)
        elif isinstance(node, list) and part.isdigit():
            index = int(part)
            if index >= len(node):
                return None
            node = node[index]
        else:
            return None

    return node if isinstance(node, dict) else None


def _caption_texts(document: dict[str, Any], item: dict[str, Any]) -> list[str]:
    captions: list[str] = []
    for reference in item.get("captions") or []:
        resolved = _resolve_ref(document, reference)
        if not resolved:
            continue
        text = resolved.get("text") or resolved.get("orig")
        if isinstance(text, str) and text.strip():
            captions.append(" ".join(text.split()))
    return captions


def _first_provenance(item: dict[str, Any]) -> dict[str, Any] | None:
    provenance = item.get("prov") or []
    if provenance and isinstance(provenance[0], dict):
        return provenance[0]
    return None


def parse_docling_document(
    document: dict[str, Any],
    *,
    picture_assets: list[dict[str, Any]] | None = None,
    table_assets: list[dict[str, Any]] | None = None,
) -> dict[str, Any]:
    picture_assets = picture_assets or []
    table_assets = table_assets or []
    picture_assets_by_id = {
        asset["element_id"]: asset
        for asset in picture_assets
        if isinstance(asset.get("element_id"), str) and asset["element_id"]
    }
    table_assets_by_id = {
        asset["element_id"]: asset
        for asset in table_assets
        if isinstance(asset.get("element_id"), str) and asset["element_id"]
    }

    formulas: list[dict[str, Any]] = []
    for item in document.get("texts") or []:
        if str(item.get("label", "")).lower() != "formula":
            continue
        formulas.append(
            {
                "id": item.get("self_ref"),
                "text": item.get("text"),
                "original": item.get("orig"),
                "provenance": item.get("prov") or [],
            }
        )

    tables: list[dict[str, Any]] = []
    table_issues: list[dict[str, Any]] = []
    for index, item in enumerate(document.get("tables") or [], start=1):
        data = item.get("data") or {}
        cells = data.get("table_cells") or []
        rows = int(data.get("num_rows") or 0)
        columns = int(data.get("num_cols") or 0)
        structure_status = (
            "structured" if rows > 0 and columns > 0 and len(cells) > 0 else "empty_structure"
        )
        item_id = item.get("self_ref") or f"table-{index}"
        parsed = {
            "id": item_id,
            "index": index,
            "captions": _caption_texts(document, item),
            "provenance": item.get("prov") or [],
            "rows": rows,
            "columns": columns,
            "cell_count": len(cells),
            "orientation": data.get("orientation"),
            "structure_status": structure_status,
            "asset": table_assets_by_id.get(item_id)
            or (table_assets[index - 1] if index <= len(table_assets) else None),
        }
        tables.append(parsed)
        if structure_status != "structured":
            table_issues.append(
                {
                    "type": "table_empty_structure",
                    "table_id": parsed["id"],
                    "index": index,
                    "provenance": parsed["provenance"],
                }
            )

    pictures: list[dict[str, Any]] = []
    for index, item in enumerate(document.get("pictures") or [], start=1):
        item_id = item.get("self_ref") or f"picture-{index}"
        pictures.append(
            {
                "id": item_id,
                "index": index,
                "captions": _caption_texts(document, item),
                "provenance": item.get("prov") or [],
                "asset": picture_assets_by_id.get(item_id)
                or (picture_assets[index - 1] if index <= len(picture_assets) else None),
            }
        )

    return {
        "formulas": formulas,
        "tables": tables,
        "pictures": pictures,
        "quality_flags": table_issues,
        "counts": {
            "formulas": len(formulas),
            "tables": len(tables),
            "tables_structured": sum(
                1 for table in tables if table["structure_status"] == "structured"
            ),
            "tables_empty_structure": len(table_issues),
            "pictures": len(pictures),
        },
    }

from typing import Any

from lxml import etree

NS = {"tei": "http://www.tei-c.org/ns/1.0"}
XML_ID = "{http://www.w3.org/XML/1998/namespace}id"


def _clean(value: str | None) -> str | None:
    if value is None:
        return None
    cleaned = " ".join(value.split())
    return cleaned or None


def _node_text(node: etree._Element | None) -> str | None:
    if node is None:
        return None
    return _clean(" ".join(node.itertext()))


def _first_text(root: etree._Element, xpath: str) -> str | None:
    nodes = root.xpath(xpath, namespaces=NS)
    if not nodes:
        return None
    first = nodes[0]
    if isinstance(first, etree._Element):
        return _node_text(first)
    return _clean(str(first))


def _all_text(root: etree._Element, xpath: str) -> list[str]:
    values: list[str] = []
    for node in root.xpath(xpath, namespaces=NS):
        text = _node_text(node) if isinstance(node, etree._Element) else _clean(str(node))
        if text:
            values.append(text)
    return values


def _coords(node: etree._Element) -> str | None:
    return node.get("coords")


def _parse_authors(root: etree._Element) -> list[dict[str, Any]]:
    author_nodes = root.xpath(
        ".//tei:teiHeader//tei:sourceDesc//tei:analytic/tei:author",
        namespaces=NS,
    )
    if not author_nodes:
        author_nodes = root.xpath(".//tei:teiHeader//tei:titleStmt/tei:author", namespaces=NS)

    authors: list[dict[str, Any]] = []
    seen: set[str] = set()
    for author in author_nodes:
        full_name = _node_text(author)
        if not full_name or full_name in seen:
            continue
        seen.add(full_name)
        authors.append(
            {
                "full_name": full_name,
                "forename": _first_text(author, ".//tei:forename[1]"),
                "surname": _first_text(author, ".//tei:surname[1]"),
                "orcid": _first_text(author, ".//tei:idno[@type='ORCID'][1]"),
                "email": _first_text(author, ".//tei:email[1]"),
                "coords": _coords(author),
            }
        )
    return authors


def _parse_sections(root: etree._Element) -> list[dict[str, Any]]:
    sections: list[dict[str, Any]] = []
    for index, div in enumerate(
        root.xpath(".//tei:text/tei:body//tei:div", namespaces=NS),
        start=1,
    ):
        heading_node = div.find("{http://www.tei-c.org/ns/1.0}head")
        paragraphs = [
            text
            for paragraph in div.xpath("./tei:p", namespaces=NS)
            if (text := _node_text(paragraph))
        ]
        sections.append(
            {
                "id": div.get(XML_ID) or f"section-{index}",
                "heading": _node_text(heading_node),
                "number": heading_node.get("n") if heading_node is not None else None,
                "paragraphs": paragraphs,
                "coords": _coords(heading_node) if heading_node is not None else None,
            }
        )
    return sections


def _parse_references(root: etree._Element) -> list[dict[str, Any]]:
    references: list[dict[str, Any]] = []
    for index, bibl in enumerate(
        root.xpath(".//tei:text/tei:back//tei:listBibl/tei:biblStruct", namespaces=NS),
        start=1,
    ):
        date_nodes = bibl.xpath(".//tei:imprint/tei:date[1]", namespaces=NS)
        year = None
        if date_nodes:
            year = date_nodes[0].get("when") or _node_text(date_nodes[0])
        references.append(
            {
                "id": bibl.get(XML_ID) or f"reference-{index}",
                "title": _first_text(bibl, ".//tei:analytic/tei:title[1]"),
                "authors": _all_text(bibl, ".//tei:analytic/tei:author/tei:persName"),
                "year": year,
                "doi": _first_text(bibl, ".//tei:idno[@type='DOI'][1]"),
                "raw": _first_text(bibl, ".//tei:note[@type='raw_reference'][1]"),
                "coords": _coords(bibl),
            }
        )
    return references


def _parse_figures(root: etree._Element) -> list[dict[str, Any]]:
    figures: list[dict[str, Any]] = []
    for index, figure in enumerate(root.xpath(".//tei:text//tei:figure", namespaces=NS), start=1):
        figures.append(
            {
                "id": figure.get(XML_ID) or f"figure-{index}",
                "type": figure.get("type") or "figure",
                "label": _first_text(figure, "./tei:head[1]"),
                "caption": _first_text(figure, "./tei:figDesc[1]"),
                "content": _first_text(figure, "./tei:table[1]"),
                "coords": _coords(figure),
            }
        )
    return figures


def _parse_formulas(root: etree._Element) -> list[dict[str, Any]]:
    formulas: list[dict[str, Any]] = []
    for index, formula in enumerate(root.xpath(".//tei:text//tei:formula", namespaces=NS), start=1):
        formulas.append(
            {
                "id": formula.get(XML_ID) or f"formula-{index}",
                "label": formula.get("n"),
                "text": _node_text(formula),
                "coords": _coords(formula),
            }
        )
    return formulas


def parse_tei(tei_xml: str) -> dict[str, Any]:
    parser = etree.XMLParser(resolve_entities=False, no_network=True, recover=False)
    root = etree.fromstring(tei_xml.encode("utf-8"), parser=parser)

    sections = _parse_sections(root)
    references = _parse_references(root)
    figures = _parse_figures(root)
    formulas = _parse_formulas(root)
    authors = _parse_authors(root)

    title = _first_text(
        root,
        ".//tei:teiHeader//tei:titleStmt/tei:title[@type='main'][1]",
    ) or _first_text(root, ".//tei:teiHeader//tei:titleStmt/tei:title[1]")

    metadata = {
        "title": title,
        "authors": authors,
        "abstract": _first_text(root, ".//tei:teiHeader//tei:profileDesc/tei:abstract[1]"),
        "doi": _first_text(root, ".//tei:teiHeader//tei:idno[@type='DOI'][1]"),
        "journal": _first_text(
            root,
            ".//tei:teiHeader//tei:sourceDesc//tei:monogr/tei:title[@level='j'][1]",
        ),
        "publication_date": _first_text(
            root,
            ".//tei:teiHeader//tei:sourceDesc//tei:imprint/tei:date[1]/@when",
        ),
        "language": root.get("{http://www.w3.org/XML/1998/namespace}lang"),
    }

    table_count = sum(1 for item in figures if item["type"] == "table")
    figure_count = len(figures) - table_count

    return {
        "metadata": metadata,
        "sections": sections,
        "references": references,
        "figures_and_tables": figures,
        "formulas": formulas,
        "counts": {
            "authors": len(authors),
            "sections": len(sections),
            "references": len(references),
            "figures": figure_count,
            "tables": table_count,
            "formulas": len(formulas),
        },
    }

import asyncio
import importlib.metadata
import logging
import shutil
import time
from pathlib import Path
from typing import Any

from starlette.concurrency import run_in_threadpool

from app.config import Settings
from app.parsers.docling import parse_docling_document
from app.utils import asset_url

logger = logging.getLogger(__name__)


class DoclingService:
    """Long-lived Docling converter guarded by a concurrency limit."""

    def __init__(self, settings: Settings) -> None:
        self.settings = settings
        self._converter: Any | None = None
        self._init_lock = asyncio.Lock()
        self._semaphore = asyncio.Semaphore(settings.docling_max_concurrency)
        self._initialization_error: str | None = None
        try:
            self.version = importlib.metadata.version("docling")
        except importlib.metadata.PackageNotFoundError:
            self.version = None

    @property
    def is_initialized(self) -> bool:
        return self._converter is not None

    @property
    def initialization_error(self) -> str | None:
        return self._initialization_error

    async def initialize(self) -> None:
        if self._converter is not None:
            return
        async with self._init_lock:
            if self._converter is not None:
                return
            try:
                self._converter = await run_in_threadpool(self._build_converter)
                self._initialization_error = None
            except Exception as exc:
                self._initialization_error = str(exc)
                raise

    def _build_converter(self) -> Any:
        from docling.datamodel.accelerator_options import AcceleratorOptions
        from docling.datamodel.base_models import InputFormat
        from docling.datamodel.pipeline_options import (
            PdfPipelineOptions,
            RapidOcrOptions,
            TableFormerMode,
        )
        from docling.document_converter import DocumentConverter, PdfFormatOption

        options = PdfPipelineOptions()
        options.accelerator_options = AcceleratorOptions(
            device=self.settings.docling_accelerator_device,
            num_threads=self.settings.docling_accelerator_num_threads,
        )
        options.document_timeout = self.settings.docling_document_timeout_seconds

        options.do_ocr = self.settings.docling_do_ocr
        if options.do_ocr:
            options.ocr_options = RapidOcrOptions(
                backend=self.settings.docling_ocr_backend,
                force_full_page_ocr=self.settings.docling_ocr_force_full_page,
                bitmap_area_threshold=self.settings.docling_ocr_bitmap_area_threshold,
            )

        options.do_table_structure = self.settings.docling_do_table_structure
        options.table_structure_options.mode = (
            TableFormerMode.ACCURATE
            if self.settings.docling_table_mode == "accurate"
            else TableFormerMode.FAST
        )
        options.table_structure_options.do_cell_matching = (
            self.settings.docling_table_do_cell_matching
        )

        options.do_formula_enrichment = self.settings.docling_do_formula_enrichment
        options.images_scale = self.settings.docling_images_scale
        options.generate_page_images = self.settings.docling_generate_page_images
        options.generate_picture_images = self.settings.docling_generate_picture_images

        return DocumentConverter(
            allowed_formats=[InputFormat.PDF],
            format_options={
                InputFormat.PDF: PdfFormatOption(pipeline_options=options),
            },
        )

    async def runtime_info(self) -> dict[str, Any]:
        return await run_in_threadpool(self._runtime_info_sync)

    def _runtime_info_sync(self) -> dict[str, Any]:
        info: dict[str, Any] = {
            "configured_device": self.settings.docling_accelerator_device,
            "ocr_backend": self.settings.docling_ocr_backend,
            "initialized": self.is_initialized,
            "initialization_error": self.initialization_error,
            "docling_version": self.version,
        }
        try:
            import torch

            cuda_available = torch.cuda.is_available()
            info.update(
                {
                    "torch_version": torch.__version__,
                    "torch_cuda_version": torch.version.cuda,
                    "cuda_available": cuda_available,
                    "cuda_device_count": torch.cuda.device_count() if cuda_available else 0,
                    "gpu_name": torch.cuda.get_device_name(0) if cuda_available else None,
                    "gpu_capability": (
                        torch.cuda.get_device_capability(0) if cuda_available else None
                    ),
                    "compiled_architectures": torch.cuda.get_arch_list() if cuda_available else [],
                }
            )
        except Exception as exc:  # pragma: no cover - diagnostics must not break health endpoints
            info["torch_error"] = str(exc)
        return info

    async def extract(
        self,
        pdf_path: Path,
        *,
        document_id: str,
        include_markdown: bool,
        export_pictures: bool,
        export_tables: bool,
    ) -> dict[str, Any]:
        await self.initialize()
        assert self._converter is not None

        async with self._semaphore:
            started = time.perf_counter()
            result = await run_in_threadpool(
                self._convert_sync,
                pdf_path,
                document_id,
                include_markdown,
                export_pictures,
                export_tables,
            )
            result["duration_seconds"] = round(time.perf_counter() - started, 3)
            return result

    def _convert_sync(
        self,
        pdf_path: Path,
        document_id: str,
        include_markdown: bool,
        export_pictures: bool,
        export_tables: bool,
    ) -> dict[str, Any]:
        assert self._converter is not None

        conversion = self._converter.convert(
            pdf_path,
            max_num_pages=self.settings.docling_max_num_pages,
            max_file_size=self.settings.max_upload_size_bytes,
        )
        document = conversion.document
        document_json = document.export_to_dict()

        picture_assets, table_assets = self._export_assets(
            document,
            document_id=document_id,
            export_pictures=export_pictures,
            export_tables=export_tables,
        )
        parsed = parse_docling_document(
            document_json,
            picture_assets=picture_assets,
            table_assets=table_assets,
        )

        counts = parsed["counts"]
        stats = {
            "pages": len(document.pages),
            "text_items": len(document.texts),
            "tables": len(document.tables),
            "pictures": len(document.pictures),
            "groups": len(document.groups),
            "formulas": counts["formulas"],
            "tables_structured": counts["tables_structured"],
            "tables_empty_structure": counts["tables_empty_structure"],
            "exported_pictures": sum(
                1 for asset in picture_assets if asset.get("status") == "saved"
            ),
            "exported_tables": sum(1 for asset in table_assets if asset.get("status") == "saved"),
        }

        raw: dict[str, Any] = {
            "document_json": document_json,
            "assets": {
                "pictures": picture_assets,
                "tables": table_assets,
            },
        }
        if include_markdown:
            raw["markdown"] = document.export_to_markdown()

        return {
            "version": self.version,
            "stats": stats,
            "raw": raw,
            "parsed": parsed,
        }

    def _export_assets(
        self,
        document: Any,
        *,
        document_id: str,
        export_pictures: bool,
        export_tables: bool,
    ) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
        from docling_core.types.doc import PictureItem, TableItem

        document_dir = self.settings.assets_dir / document_id
        pictures_dir = document_dir / "figures"
        tables_dir = document_dir / "tables"

        # Reprocessing the same content hash should not leave stale files from an
        # earlier run with a different number of elements.
        if export_pictures and pictures_dir.exists():
            shutil.rmtree(pictures_dir)
        if export_tables and tables_dir.exists():
            shutil.rmtree(tables_dir)

        if export_pictures:
            pictures_dir.mkdir(parents=True, exist_ok=True)
        if export_tables:
            tables_dir.mkdir(parents=True, exist_ok=True)

        picture_assets: list[dict[str, Any]] = []
        table_assets: list[dict[str, Any]] = []
        picture_index = 0
        table_index = 0

        for element, _level in document.iterate_items():
            if export_tables and isinstance(element, TableItem):
                table_index += 1
                asset = self._save_element_image(
                    element,
                    document,
                    output_dir=tables_dir,
                    relative_dir=Path(document_id) / "tables",
                    prefix="table",
                    index=table_index,
                )
                table_assets.append(asset)

            if export_pictures and isinstance(element, PictureItem):
                picture_index += 1
                asset = self._save_element_image(
                    element,
                    document,
                    output_dir=pictures_dir,
                    relative_dir=Path(document_id) / "figures",
                    prefix="figure",
                    index=picture_index,
                )
                picture_assets.append(asset)

        return picture_assets, table_assets

    def _save_element_image(
        self,
        element: Any,
        document: Any,
        *,
        output_dir: Path,
        relative_dir: Path,
        prefix: str,
        index: int,
    ) -> dict[str, Any]:
        image = element.get_image(document)
        element_id = str(getattr(element, "self_ref", "")) or None
        if image is None:
            logger.warning("Docling returned no image for %s %s", prefix, index)
            return {
                "index": index,
                "element_id": element_id,
                "status": "not_available",
                "error": "Docling returned no image for this element.",
            }

        filename = f"{prefix}_{index:03d}.png"
        path = output_dir / filename
        with path.open("wb") as file_handle:
            image.save(file_handle, format="PNG", compress_level=6)

        relative_path = (relative_dir / filename).as_posix()
        return {
            "index": index,
            "element_id": element_id,
            "status": "saved",
            "filename": filename,
            "relative_path": relative_path,
            "url": asset_url(self.settings.assets_url_prefix, relative_path),
            "width": image.width,
            "height": image.height,
            "size_bytes": path.stat().st_size,
        }

import asyncio
import time
from pathlib import Path
from typing import Any

import httpx

from app.config import Settings
from app.parsers.tei import parse_tei


class GrobidServiceError(RuntimeError):
    pass


class GrobidService:
    def __init__(self, settings: Settings) -> None:
        self.settings = settings
        self._client = httpx.AsyncClient(
            base_url=settings.grobid_base_url,
            timeout=httpx.Timeout(settings.grobid_timeout_seconds),
            headers={"Accept": "application/xml"},
        )
        self._version: str | None = None

    async def close(self) -> None:
        await self._client.aclose()

    async def is_alive(self) -> bool:
        try:
            response = await self._client.get("/api/isalive", timeout=10.0)
            return response.status_code == 200 and response.text.strip().lower() == "true"
        except httpx.HTTPError:
            return False

    async def version(self) -> str | None:
        if self._version is not None:
            return self._version
        try:
            response = await self._client.get("/api/version", timeout=10.0)
            response.raise_for_status()
            self._version = response.text.strip()
        except httpx.HTTPError:
            self._version = None
        return self._version

    async def extract(
        self,
        pdf_path: Path,
        *,
        filename: str,
        consolidate_header: str,
        consolidate_citations: str,
        include_raw_citations: bool,
        include_raw_affiliations: bool,
        segment_sentences: bool,
        include_tei_xml: bool,
    ) -> dict[str, Any]:
        started = time.perf_counter()
        tei_xml = await self._process_fulltext(
            pdf_path,
            filename=filename,
            consolidate_header=consolidate_header,
            consolidate_citations=consolidate_citations,
            include_raw_citations=include_raw_citations,
            include_raw_affiliations=include_raw_affiliations,
            segment_sentences=segment_sentences,
        )
        parsed = parse_tei(tei_xml)

        return {
            "version": await self.version(),
            "duration_seconds": round(time.perf_counter() - started, 3),
            "stats": parsed.get("counts", {}),
            "raw": {"tei_xml": tei_xml} if include_tei_xml else None,
            "parsed": parsed,
        }

    async def _process_fulltext(
        self,
        pdf_path: Path,
        *,
        filename: str,
        consolidate_header: str,
        consolidate_citations: str,
        include_raw_citations: bool,
        include_raw_affiliations: bool,
        segment_sentences: bool,
    ) -> str:
        text_fields: list[tuple[str, str]] = [
            ("consolidateHeader", consolidate_header),
            ("consolidateCitations", consolidate_citations),
            ("includeRawCitations", "1" if include_raw_citations else "0"),
            ("includeRawAffiliations", "1" if include_raw_affiliations else "0"),
            ("segmentSentences", "1" if segment_sentences else "0"),
            ("generateIDs", "1"),
        ]
        text_fields.extend(
            ("teiCoordinates", coordinate) for coordinate in self.settings.grobid_tei_coordinates
        )

        last_error: Exception | None = None
        for attempt in range(1, self.settings.grobid_max_retries + 1):
            try:
                with pdf_path.open("rb") as pdf_file:
                    # Put both the file and normal fields in one multipart list so
                    # repeated `teiCoordinates` values are preserved reliably.
                    multipart: list[tuple[str, tuple[str | None, Any, str | None]]] = [
                        ("input", (filename, pdf_file, "application/pdf")),
                    ]
                    multipart.extend((name, (None, value, None)) for name, value in text_fields)
                    response = await self._client.post(
                        "/api/processFulltextDocument",
                        files=multipart,
                    )

                if response.status_code in {429, 502, 503, 504}:
                    raise GrobidServiceError(
                        f"GROBID returned retryable HTTP {response.status_code}."
                    )
                if response.status_code == 204:
                    raise GrobidServiceError("GROBID completed but extracted no content.")

                response.raise_for_status()
                if not response.text.strip():
                    raise GrobidServiceError("GROBID returned an empty response.")
                return response.text
            except (httpx.HTTPError, GrobidServiceError) as exc:
                last_error = exc
                if attempt < self.settings.grobid_max_retries:
                    await asyncio.sleep(min(2**attempt, 10))

        raise GrobidServiceError(f"GROBID request failed: {last_error}")

        "note": (

            "quality scoring, and agentic repair belong in the next pipeline stage."
        ),
    }

import hashlib
import os
import tempfile
from pathlib import Path

from fastapi import HTTPException, UploadFile, status

PDF_MAGIC = b"%PDF-"


async def save_upload_to_temp(
    upload: UploadFile,
    *,
    max_bytes: int,
    chunk_size: int = 1024 * 1024,
) -> tuple[Path, str, int]:
    """Stream an uploaded PDF to disk while hashing it and enforcing a size limit."""
    digest = hashlib.sha256()
    total = 0
    first_chunk = True

    fd, temp_name = tempfile.mkstemp(prefix="pdf-extract-", suffix=".pdf")
    os.close(fd)
    temp_path = Path(temp_name)

    try:
        with temp_path.open("wb") as output:
            while chunk := await upload.read(chunk_size):
                if first_chunk:
                    first_chunk = False
                    if not chunk.startswith(PDF_MAGIC):
                        raise HTTPException(
                            status_code=status.HTTP_415_UNSUPPORTED_MEDIA_TYPE,
                            detail="The uploaded file does not have a valid PDF signature.",
                        )

                total += len(chunk)
                if total > max_bytes:
                    raise HTTPException(
                        status_code=status.HTTP_413_REQUEST_ENTITY_TOO_LARGE,
                        detail=f"PDF exceeds the configured limit of {max_bytes} bytes.",
                    )

                digest.update(chunk)
                output.write(chunk)

        if total == 0:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="The uploaded PDF is empty.",
            )

        return temp_path, digest.hexdigest(), total
    except Exception:
        temp_path.unlink(missing_ok=True)
        raise
    finally:
        await upload.close()


def safe_filename(name: str | None) -> str:
    return Path(name).name if name else "document.pdf"


def asset_url(prefix: str, relative_path: str) -> str:
    return f"{prefix.rstrip('/')}/{relative_path.lstrip('/')}"

#!/usr/bin/env python3
import argparse
import json
import sys
import time
import urllib.error
import urllib.request


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("url")
    parser.add_argument("--timeout", type=int, default=600)
    parser.add_argument("--interval", type=float, default=5.0)
    args = parser.parse_args()

    deadline = time.monotonic() + args.timeout
    last_error: str | None = None
    while time.monotonic() < deadline:
        try:
            with urllib.request.urlopen(args.url, timeout=10) as response:
                body = response.read().decode("utf-8", errors="replace")
                if 200 <= response.status < 300:
                    print(body)
                    return 0
        except (urllib.error.URLError, TimeoutError) as exc:
            last_error = str(exc)
        time.sleep(args.interval)

    print(json.dumps({"status": "timeout", "url": args.url, "last_error": last_error}))
    return 1


if __name__ == "__main__":
    sys.exit(main())
